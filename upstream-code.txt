# 1.qc raw-data
mkdir -p 2.QC-data/raw-qc 2.QC-data/clean-qc
fastqc -t 12 -q -o 2.QC-data/raw-qc/ 1.raw-data/*.gz
multiqc 2.QC-data/raw-qc/* -o 2.QC-data/raw-qc/

# 2.trim adaptors
mkdir 3.trim-data

for i in SRR74712{27..44} SRR74712{49..60};do cutadapt -j 10 -f fastq -a CTGTAGGCACCATCAATTCGTATGCCGTCTT -O 6 -m 20 -M 35 -o 3.trim-data/${i}.trimmed.fq.gz 1.raw-data/${i}.fastq.gz;done

fastqc -t 12 -q -o 2.QC-data/clean-qc/ 3.trim-data/*.gz
multiqc 2.QC-data/clean-qc/* -o 2.QC-data/clean-qc/

# 3.build index for rRNA/tRNA and genome
mkdir rRNA-index tRNA-index genome-index bw1-genome-index

bowtie2-build Saccharomyces-cerevisiae-rRNA.fasta rRNA-index/Saccharomyces-cerevisiae-rRNA
bowtie2-build Saccharomyces-cerevisiae-tRNA.fasta tRNA-index/Saccharomyces-cerevisiae-tRNA

# 4.rm rtRNA contamination from fastq files
mkdir 4.rmrtRNA-data

for i in SRR74712{27..44} SRR74712{49..60};do bowtie2 -p 20 -x ./index/rRNA-index/Saccharomyces-cerevisiae-rRNA --un-gz 4.rmrtRNA-data/${i}.rmrRNA.fq.gz -U 3.trim-data/${i}.trimmed.fq.gz -S 4.rmrtRNA-data/null;done

for i in SRR74712{27..44} SRR74712{49..60};do bowtie2 -p 20 -x ./index/tRNA-index/Saccharomyces-cerevisiae-tRNA --un-gz 4.rmrtRNA-data/${i}.rmtRNA.fq.gz -U 4.rmrtRNA-data/${i}.rmrRNA.fq.gz -S 4.rmrtRNA-data/null;done

# 5.map to trancriptome
mkdir trans-index
mkdir 6.map-trans-data

hisat2-build extened-CDS-seq.fa trans-index/extened-CDS-seq

for i in SRR74712{27..44} SRR74712{49..60};do bowtie -p 20 ./index/trans-index/extened-CDS-seq -m 1 -v 2 --best --strata <(zcat 4.rmrtRNA-data/${i}.rmtRNA.fq.gz) 6.map-data/${i}.map;done

for i in SRR74712{27..44} SRR74712{49..60};do hisat2 -p 20 -x ./index/trans-index/extened-CDS-seq -k 1 --summary-file 5.map-data/${i}.mapinfo.txt -U 4.rmrtRNA-data/${i}.rmtRNA.fq.gz -S 5.map-data/${i}.sam;done
